/**
 * Migration 003: Lock down tables for hosted Postgres (Supabase)
 *
 * Supabase publishes every table in `public` through its auto-generated REST API
 * to the `anon` and `authenticated` roles, and the anon key is public by design.
 * Without this, anyone holding a project's anon key could read `users` (password
 * hashes) and every user's library. Folio never uses that API: the Fastify server
 * is the only client. So we enable RLS with no policies (deny all) and revoke
 * those roles' privileges, including on tables created by future migrations.
 *
 * The server connects as the tables' owner, which bypasses RLS, so this is a
 * no-op for the app itself and for plain Postgres (Docker) deployments.
 * New tables added later should also `enable row level security`.
 */
export const up = (pgm) => {
  pgm.sql(`
    do $$
    declare
      t text;
      r text;
    begin
      for t in select tablename from pg_tables where schemaname = 'public' loop
        execute format('alter table public.%I enable row level security', t);
      end loop;

      foreach r in array array['anon', 'authenticated'] loop
        if exists (select 1 from pg_roles where rolname = r) then
          execute format('revoke all on all tables in schema public from %I', r);
          execute format('revoke all on all sequences in schema public from %I', r);
          execute format('revoke all on all functions in schema public from %I', r);
          execute format('alter default privileges in schema public revoke all on tables from %I', r);
          execute format('alter default privileges in schema public revoke all on sequences from %I', r);
          execute format('alter default privileges in schema public revoke all on functions from %I', r);
        end if;
      end loop;
    end $$;
  `);
};

 *   their sync cursor and uploaded-file cache by it, so restoring, wiping or
 *   replacing the database (or pointing the app at a different server) makes
 *   every client re-push and re-pull instead of silently skipping data because
 *   its cursor is ahead of the new sequence.
 * - deleted_users: tombstones for deleted accounts (id + time only, no personal

import crypto from 'node:crypto';
import type { PoolClient } from 'pg';
import { pool } from './db.js';

let instanceId: string | null = null;

/** Random id of this server database (see migration 004). Cached after first read. */
export async function getInstanceId(): Promise<string> {
  if (instanceId) return instanceId;
  try {
    const r = await pool.query<{ instance_id: string }>('select instance_id from server_meta limit 1');
    if (r.rows[0]?.instance_id) instanceId = String(r.rows[0].instance_id);
  } catch {
    // Table missing (migrations not applied yet): fall through.
  }
  // Without the table the id is per-process; clients then resync after a restart,
  // which is safe (just redundant). Migrations normally create it.
  instanceId ??= crypto.randomUUID();
  return instanceId;
}

// Whether a user id still exists. Access tokens outlive account deletion by up
// to their TTL, so authenticated routes check this (cached briefly).
const existsCache = new Map<string, { exists: boolean; at: number }>();
const EXISTS_TTL_MS = 30_000;

export async function userExists(userId: string): Promise<boolean> {
  const hit = existsCache.get(userId);
  if (hit && Date.now() - hit.at < EXISTS_TTL_MS) return hit.exists;
  const r = await pool.query('select 1 from users where id = $1', [userId]);
  const exists = (r.rowCount ?? 0) > 0;
  existsCache.set(userId, { exists, at: Date.now() });
  return exists;
}

export function forgetUser(userId: string): void {
  existsCache.set(userId, { exists: false, at: Date.now() });
}

/** True if this id belonged to an account that was deleted. */
export async function wasDeleted(userId: string): Promise<boolean> {
  try {
    const r = await pool.query('select 1 from deleted_users where id = $1', [userId]);
    return (r.rowCount ?? 0) > 0;
  } catch {
    return false;
  }
}

/**
 * Serialises one user's sync writes for the rest of the transaction.
 *
 * server_seq values are taken from a sequence when a row is written, not when
 * the transaction commits. With two devices pushing at once, a pull could see
 * the later transaction's rows (higher seq) first, advance its cursor past the
 * earlier transaction's still-uncommitted rows, and never receive them. Holding
 * a per-user lock for the whole push makes one user's writes commit in seq order.
 */
export async function lockUserWrites(client: PoolClient, userId: string): Promise<void> {
  // 48-bit key from the user id keeps us inside JS's safe-integer range.
  const key = parseInt(crypto.createHash('sha256').update(userId).digest('hex').slice(0, 12), 16);
  await client.query('select pg_advisory_xact_lock($1)', [key]);
}

import { fileURLToPath, pathToFileURL } from 'node:url';
import { runner } from 'node-pg-migrate';
import { buildPoolConfig } from './db.js';

const MIGRATIONS_DIR = fileURLToPath(new URL('../migrations', import.meta.url));

/**
 * Applies pending migrations using the same connection settings as the API
 * (including Supabase SSL handling), so it works anywhere the server does.
 */
export async function runMigrations(direction: 'up' | 'down' = 'up', count?: number): Promise<void> {
  const { connectionString, ssl } = buildPoolConfig();
  if (!connectionString) throw new Error('DATABASE_URL is not set');
  const applied = await runner({
    databaseUrl: { connectionString, ssl },
    dir: MIGRATIONS_DIR,
    migrationsTable: 'pgmigrations',
    direction,
    count: count ?? (direction === 'up' ? Infinity : 1),
    singleTransaction: true,
    log: (msg) => console.log(msg),
  });
  console.log(
    applied.length
      ? `Migrations ${direction}: ${applied.map((m) => m.name).join(', ')}`
      : 'Database schema is up to date.',
  );
}

// `node dist/migrate.js [up|down] [count]`
if (process.argv[1] && import.meta.url === pathToFileURL(process.argv[1]).href) {
  const direction = process.argv[2] === 'down' ? 'down' : 'up';
  const count = process.argv[3] ? Number(process.argv[3]) : undefined;
  runMigrations(direction, count).catch((err) => {
    console.error('Migration failed:', err instanceof Error ? err.message : err);
    process.exit(1);
  });
}

import pg from 'pg';
import { buildPoolConfig } from './db.js';

type Handler = () => void;

const MAX_RETRY_MS = 30_000;

function quoteIdent(name: string): string {
  return `"${name.replace(/"/g, '""')}"`;
}

function defaultClient(): pg.Client {
  const { connectionString, ssl, keepAlive, connectionTimeoutMillis } = buildPoolConfig();
  return new pg.Client({ connectionString, ssl, keepAlive, connectionTimeoutMillis });
}

/**
 * Multiplexes Postgres LISTEN/NOTIFY over a single dedicated connection.
 *
 * Each SSE stream used to check a client out of the pool and hold it for as long
 * as the device stayed connected, so a handful of open devices exhausted the pool
 * (and a hosted database's connection limit) and stalled every other request.
 * Now every stream shares one connection; channels are LISTENed while at least
 * one subscriber needs them. If the connection drops it is re-established with
 * backoff, all channels are re-LISTENed, and every subscriber is pinged once so
 * clients re-pull anything they missed while it was down.
 */
export class NotifyHub {
  private client: pg.Client | null = null;
  private connecting: Promise<pg.Client> | null = null;
  private handlers = new Map<string, Set<Handler>>();
  /** Channels LISTENed on the current connection (reset when it is replaced). */
  private listening = new Set<string>();
  private retryTimer: NodeJS.Timeout | null = null;
  private retryMs = 1000;
  private closed = false;

  constructor(private readonly createClient: () => pg.Client = defaultClient) {}

  /** Registers `handler` for `channel`. Resolves to an unsubscribe function. */
  async subscribe(channel: string, handler: Handler): Promise<() => Promise<void>> {
    let set = this.handlers.get(channel);
    const firstForChannel = !set;
    if (!set) {
      set = new Set();
      this.handlers.set(channel, set);
    }
    set.add(handler);

    if (firstForChannel) {
      try {
        await this.listen(await this.ensureClient(), channel);
      } catch (err) {
        // The reconnect loop LISTENs on every registered channel once it's back.
        console.error('LISTEN failed; will retry:', (err as Error).message);
        this.scheduleReconnect();
      }
    }

    return async () => {
      const current = this.handlers.get(channel);
      if (!current) return;
      current.delete(handler);
      if (current.size > 0) return;
      this.handlers.delete(channel);
      if (this.client && this.listening.delete(channel)) {
        await this.client.query(`UNLISTEN ${quoteIdent(channel)}`).catch(() => {});
      }
    };
  }

  /** Number of channels currently listened to (for tests/diagnostics). */
  get channelCount(): number {
    return this.handlers.size;
  }

  async close(): Promise<void> {
    this.closed = true;
    if (this.retryTimer) clearTimeout(this.retryTimer);
    this.handlers.clear();
    const client = this.client;
    this.client = null;
    if (client) await client.end().catch(() => {});
  }

  private ensureClient(): Promise<pg.Client> {
    if (this.client) return Promise.resolve(this.client);
    if (!this.connecting) {
      this.connecting = this.connect().finally(() => {
        this.connecting = null;
      });
    }
    return this.connecting;
  }

  private async connect(): Promise<pg.Client> {
    const client = this.createClient();
    client.on('notification', (msg) => {
      this.handlers.get(msg.channel)?.forEach((h) => h());
    });
    const onLost = () => this.handleLost(client);
    client.on('error', onLost);
    client.on('end', onLost);

    await client.connect();
    if (this.closed) {
      await client.end().catch(() => {});
      throw new Error('NotifyHub closed');
    }
    this.listening = new Set();
    for (const channel of this.handlers.keys()) {
      await this.listen(client, channel);
    }
    this.client = client;
    this.retryMs = 1000;
    return client;
  }

  private async listen(client: pg.Client, channel: string): Promise<void> {
    if (this.listening.has(channel)) return;
    this.listening.add(channel);
    try {
      await client.query(`LISTEN ${quoteIdent(channel)}`);
    } catch (err) {
      this.listening.delete(channel);
      throw err;
    }
  }

  private handleLost(client: pg.Client): void {
    if (this.client !== client) return;
    this.client = null;
    client.end().catch(() => {});
    this.scheduleReconnect();
  }

  private scheduleReconnect(): void {
    if (this.closed || this.retryTimer || this.handlers.size === 0) return;
    const delay = this.retryMs;
    this.retryMs = Math.min(this.retryMs * 2, MAX_RETRY_MS);
    this.retryTimer = setTimeout(() => {
      this.retryTimer = null;
      if (this.closed || this.handlers.size === 0) return;
      this.ensureClient()
        .then(() => {
          // Changes made while we were disconnected produced no notifications.
          for (const set of this.handlers.values()) set.forEach((h) => h());
        })
        .catch((err) => {
          console.error('Sync listener reconnect failed:', (err as Error).message);
          this.scheduleReconnect();
        });
    }, delay);
  }
}

let hub: NotifyHub | null = null;

export function getNotifyHub(): NotifyHub {
  if (!hub) hub = new NotifyHub();
  return hub;
}

export function setNotifyHub(next: NotifyHub | null): void {
  hub = next;
}

import { EventEmitter } from 'node:events';
import { describe, it, expect, vi, afterEach } from 'vitest';
import type pg from 'pg';
import { NotifyHub } from '../src/notify.js';

class FakeClient extends EventEmitter {
  queries: string[] = [];
  connected = false;
  async connect() {
    this.connected = true;
  }
  async query(sql: string) {
    this.queries.push(sql);
    return { rows: [] };
  }
  async end() {
    this.connected = false;
  }
  notify(channel: string) {
    this.emit('notification', { channel, processId: 1, payload: '' });
  }
}

function makeHub() {
  const clients: FakeClient[] = [];
  const hub = new NotifyHub(() => {
    const c = new FakeClient();
    clients.push(c);
    return c as unknown as pg.Client;
  });
  return { hub, clients };
}

describe('NotifyHub', () => {
  afterEach(() => {
    vi.useRealTimers();
  });

  it('shares one connection and LISTENs once per channel', async () => {
    const { hub, clients } = makeHub();
    const a = vi.fn();
    const b = vi.fn();
    await hub.subscribe('sync_u1', a);
    await hub.subscribe('sync_u1', b);
    await hub.subscribe('sync_u2', vi.fn());

    expect(clients).toHaveLength(1);
    expect(clients[0].queries).toEqual(['LISTEN "sync_u1"', 'LISTEN "sync_u2"']);

    clients[0].notify('sync_u1');
    expect(a).toHaveBeenCalledTimes(1);
    expect(b).toHaveBeenCalledTimes(1);
    await hub.close();
  });

  it('only UNLISTENs when the last subscriber leaves', async () => {
    const { hub, clients } = makeHub();
    const offA = await hub.subscribe('sync_u1', vi.fn());
    const offB = await hub.subscribe('sync_u1', vi.fn());

    await offA();
    expect(clients[0].queries).not.toContain('UNLISTEN "sync_u1"');
    await offB();
    expect(clients[0].queries).toContain('UNLISTEN "sync_u1"');
    expect(hub.channelCount).toBe(0);
    await hub.close();
  });

  it('does not deliver notifications for other channels', async () => {
    const { hub, clients } = makeHub();
    const handler = vi.fn();
    await hub.subscribe('sync_u1', handler);
    clients[0].notify('sync_u2');
    expect(handler).not.toHaveBeenCalled();
    await hub.close();
  });

  it('reconnects, re-LISTENs and pings subscribers after the connection drops', async () => {
    vi.useFakeTimers();
    const { hub, clients } = makeHub();
    const handler = vi.fn();
    await hub.subscribe('sync_u1', handler);

    clients[0].emit('error', new Error('connection terminated'));
    await vi.advanceTimersByTimeAsync(1000);

    expect(clients).toHaveLength(2);
    expect(clients[1].queries).toEqual(['LISTEN "sync_u1"']);
    // Subscribers re-pull, since changes during the outage produced no NOTIFY.
    expect(handler).toHaveBeenCalledTimes(1);

    clients[1].notify('sync_u1');
    expect(handler).toHaveBeenCalledTimes(2);
    await hub.close();
  });

  it('quotes channel names safely', async () => {
    const { hub, clients } = makeHub();
    await hub.subscribe('sync_a"b', vi.fn());
    expect(clients[0].queries).toEqual(['LISTEN "sync_a""b"']);
    await hub.close();
  });
});

import 'dotenv/config';
import fs from 'node:fs';
import pg from 'pg';

// Return BIGINT columns as JS numbers (ms timestamps and seqs fit safely within Number.MAX_SAFE_INTEGER)
pg.types.setTypeParser(20, (v: string) => Number(v));

type SslMode = 'disable' | 'require' | 'verify-full';

/** Hosted Postgres providers that only accept TLS connections. */
function isHostedRequiringTls(host: string): boolean {
  return /(^|\.)supabase\.(co|com)$/i.test(host);
}

function readCa(value: string | undefined): string | undefined {
  if (!value) return undefined;
  if (value.includes('-----BEGIN CERTIFICATE-----')) return value.replace(/\\n/g, '\n');
  return fs.readFileSync(value, 'utf8');
}

/**
 * Builds the pg connection config from env.
 *
 * - DATABASE_URL       postgres://user:pass@host:port/db (required outside tests)
 * - DATABASE_SSL       disable | require | verify-full. Default: `require` for Supabase
 *                      hosts or when the URL asks for sslmode, otherwise `disable`.
 * - DATABASE_CA_CERT   CA bundle (file path or PEM). With it, `require` upgrades to
 *                      full verification. Supabase's CA is downloadable from
 *                      Project Settings → Database → SSL Configuration.
 * - DATABASE_POOL_MAX  pool size (default 10). Keep it below the provider's limit
 *                      minus one: the sync event listener holds one extra connection.
 */
export function buildPoolConfig(env: NodeJS.ProcessEnv = process.env): pg.PoolConfig {
  const raw = env.DATABASE_URL;
  const max = Number(env.DATABASE_POOL_MAX) || 10;
  const base: pg.PoolConfig = {
    max,
    idleTimeoutMillis: 30_000,
    connectionTimeoutMillis: 10_000,
    // Hosted poolers drop silent TCP connections; keepalives stop that killing idle clients.
    keepAlive: true,
  };
  if (!raw) return base;

  let url: URL;
  try {
    url = new URL(raw);
  } catch {
    return { ...base, connectionString: raw };
  }

  // sslmode in the URL would override the `ssl` object below (pg lets the connection
  // string win), and pg treats `require` as full verification, which fails against
  // Supabase's private CA. Resolve it here and strip it from the URL.
  const urlMode = url.searchParams.get('sslmode');
  for (const p of ['sslmode', 'sslrootcert', 'sslcert', 'sslkey', 'uselibpqcompat']) {
    url.searchParams.delete(p);
  }

  const explicit = env.DATABASE_SSL as SslMode | undefined;
  const ca = readCa(env.DATABASE_CA_CERT);
  let mode: SslMode;
  if (explicit) mode = explicit;
  else if (urlMode === 'disable') mode = 'disable';
  else if (urlMode === 'verify-full' || urlMode === 'verify-ca') mode = 'verify-full';
  else if (urlMode || isHostedRequiringTls(url.hostname)) mode = 'require';
  else mode = 'disable';

  let ssl: pg.PoolConfig['ssl'] = false;
  if (mode === 'verify-full' || (mode === 'require' && ca)) {
    ssl = { rejectUnauthorized: true, ...(ca ? { ca } : {}) };
  } else if (mode === 'require') {
    // Encrypted, but the server certificate isn't verified against a CA.
    ssl = { rejectUnauthorized: false };
  }

  return { ...base, connectionString: url.toString(), ssl };
}

/**
 * Supabase's transaction pooler (port 6543) multiplexes clients per transaction,
 * so LISTEN/NOTIFY (used for live sync) silently never fires. Session mode works.
 */
export function describeConnection(env: NodeJS.ProcessEnv = process.env): string[] {
  const warnings: string[] = [];
  try {
    const url = new URL(env.DATABASE_URL ?? '');
    if (isHostedRequiringTls(url.hostname) && url.port === '6543') {
      warnings.push(
        'DATABASE_URL points at the Supabase transaction pooler (port 6543). Live sync uses ' +
          'LISTEN/NOTIFY, which needs a session connection: use the session pooler URI (port 5432).',
      );
    }
  } catch {
    // Unparseable URL: pg will report it on first connect.
  }
  return warnings;
}

function createPool(): pg.Pool {
  const p = new pg.Pool(buildPoolConfig());
  // An idle client can be dropped by the server or a pooler at any time (restarts,
  // maintenance). Without a listener, pg re-throws that as an uncaught error and
  // the whole API process exits.
  p.on('error', (err) => {
    console.error('Postgres idle client error:', err.message);
  });
  return p;
}

export let pool: pg.Pool = createPool();

export function setPool(newPool: pg.Pool): void {
  pool = newPool;
}

import './types.js';

const TMP_DIR = path.join(process.env.FILES_DIR || './data/files', '.tmp');
const MAX_BYTES = (Number(process.env.MAX_FILE_MB) || 300) * 1024 * 1024;

export { type FileItem };

export const fileRoutes: FastifyPluginAsync = async (app: FastifyInstance) => {
  // Pass raw upload stream to route instead of buffering in memory
  app.addContentTypeParser('application/octet-stream', (_req, payload, done) => {
    done(null, payload);
  });

  // List all files stored for the authenticated user
  app.get(
    '/',
    { preHandler: app.auth },
    async (req): Promise<FileItem[]> => {
      return storage.list(req.user.sub);
    },
  );

  app.put<{ Params: { hash: string } }>(
    '/:hash',
    { preHandler: app.auth },
    async (req, reply) => {
      const { hash } = req.params;
      if (!HASH_RE.test(hash)) {
        return reply.code(400).send({ error: 'invalid hash' });
      }

      const bodyStream = req.body as NodeJS.ReadableStream | undefined;
      if (!bodyStream || typeof (bodyStream as NodeJS.ReadableStream).pipe !== 'function') {
        return reply.code(415).send({ error: 'send body as application/octet-stream' });
      }

      if (await storage.has(req.user.sub, hash)) {
        (bodyStream as NodeJS.ReadableStream & { resume?: () => void }).resume?.();
        return { stored: false, exists: true };
      }

      await fsp.mkdir(TMP_DIR, { recursive: true });
      const tmp = path.join(TMP_DIR, `${req.user.sub}_${hash}_${crypto.randomUUID()}.part`);
      const hasher = crypto.createHash('sha256');
      let size = 0;
      const meter = new Transform({
        transform(chunk: Buffer, _enc, cb) {
          size += chunk.length;
          if (size > MAX_BYTES) {
            return cb(new Error('FILE_TOO_LARGE'));
          }
          hasher.update(chunk);
          cb(null, chunk);
        },
      });

      try {
        await pipeline(bodyStream, meter, fs.createWriteStream(tmp));
      } catch (e: unknown) {
        await fsp.rm(tmp, { force: true });
        const err = e as Error;
        if (err.message === 'FILE_TOO_LARGE') {
          return reply.code(413).send({ error: 'file too large' });
        }
        throw e;
      }

      if (hasher.digest('hex') !== hash) {
        await fsp.rm(tmp, { force: true });
        return reply.code(400).send({ error: 'hash mismatch' });
      }

      await storage.put(req.user.sub, hash, tmp, size);
      return reply.code(201).send({ stored: true, size });
    },
  );

  app.get<{ Params: { hash: string } }>(
    '/:hash',
    { preHandler: app.auth },
    async (req, reply) => {
      const { hash } = req.params;
      if (!HASH_RE.test(hash)) {
        return reply.code(400).send({ error: 'invalid hash' });
      }

      const item = await storage.get(req.user.sub, hash);
      if (!item) {
        return reply.code(404).send({ error: 'not found' });
      }

      reply
        .header('content-length', item.size)
        .header('cache-control', 'private, max-age=31536000, immutable')
        .type('application/octet-stream');

      return reply.send(item.stream);
    },
  );

  app.delete<{ Params: { hash: string } }>(
    '/:hash',
    { preHandler: app.auth },
    async (req, reply) => {
      const { hash } = req.params;
      if (!HASH_RE.test(hash)) {
        return reply.code(400).send({ error: 'invalid hash' });
      }

      await storage.delete(req.user.sub, hash);
      return { ok: true };
    },
  );
};

export default fileRoutes;

import bcrypt from 'bcryptjs';
import crypto from 'node:crypto';
import type { FastifyInstance, FastifyPluginAsync } from 'fastify';
import { pool } from './db.js';
import { wasDeleted } from './integrity.js';
import { ACCOUNT_DELETED } from '@verso/shared';
import type { AuthResponse } from '@verso/shared';
import './types.js';

const ACCESS_TTL = '1h';
const REFRESH_DAYS = 90;
const sha256 = (s: string) => crypto.createHash('sha256').update(s).digest('hex');
const strict = process.env.NODE_ENV === 'test' ? {} : { rateLimit: { max: 10, timeWindow: '1 minute' } };

const credSchema = {
  body: {
    type: 'object',
    required: ['email', 'password'],
    properties: {
      email: { type: 'string', format: 'email', maxLength: 254 },
      password: { type: 'string', minLength: 8, maxLength: 200 },
      deviceName: { type: 'string', maxLength: 100 },
    },
  },
};

export const authRoutes: FastifyPluginAsync = async (app: FastifyInstance) => {
  async function issueTokens(userId: string, email: string, deviceName?: string | null): Promise<AuthResponse> {
    const accessToken = app.jwt.sign({ sub: userId, email }, { expiresIn: ACCESS_TTL });
    const refreshToken = crypto.randomBytes(48).toString('base64url');
    const expiresAt = new Date(Date.now() + REFRESH_DAYS * 24 * 60 * 60 * 1000);
    await pool.query(
      `insert into refresh_tokens (token_hash, user_id, device_name, expires_at)
       values ($1, $2, $3, $4)`,
      [sha256(refreshToken), userId, deviceName ?? null, expiresAt],
    );
    return { accessToken, refreshToken, user: { id: userId, email } };
  }

  app.post<{ Body: { email: string; password: string; deviceName?: string } }>(
    '/register',
    { schema: credSchema, config: strict },
    async (req, reply) => {
      if (process.env.ALLOW_SIGNUP === 'false') {
        return reply.code(403).send({ error: 'signups disabled' });
      }
      const { email, password, deviceName } = req.body;
      const hash = await bcrypt.hash(password, 12);
      try {
        const r = await pool.query<{ id: string; email: string }>(
          'insert into users (email, password_hash) values ($1, $2) returning id, email',
          [email, hash],
        );
        const user = r.rows[0];
        const tokens = await issueTokens(user.id, user.email, deviceName);
        return reply.code(201).send(tokens);
      } catch (err: unknown) {
        const pgErr = err as { code?: string };
        if (pgErr.code === '23505') {
          return reply.code(409).send({ error: 'email already registered' });
        }
        throw err;
      }
    },
  );

  app.post<{ Body: { email: string; password: string; deviceName?: string } }>(
    '/login',
    { schema: credSchema, config: strict },
    async (req, reply) => {
      const { email, password, deviceName } = req.body;
      const r = await pool.query<{ id: string; email: string; password_hash: string }>(
        'select id, email, password_hash from users where email = $1',
        [email],
      );
      const user = r.rows[0];
      const ok = user && (await bcrypt.compare(password, user.password_hash));
      if (!ok) {
        return reply.code(401).send({ error: 'invalid email or password' });
      }
      return issueTokens(user.id, user.email, deviceName);
    },
  );

  app.post<{ Body: { refreshToken?: string; userId?: string } }>(
    '/refresh',
    { config: strict },
    async (req, reply) => {
      const token = req.body?.refreshToken;
      if (typeof token !== 'string') {
        return reply.code(400).send({ error: 'refreshToken required' });
      }
      // Rotation: old token is deleted atomically
      const r = await pool.query<{ user_id: string; device_name: string | null }>(
        `delete from refresh_tokens
         where token_hash = $1 and expires_at > now()
         returning user_id, device_name`,
        [sha256(token)],
      );
      if (!r.rowCount || !r.rows[0]) {
        // Deleting an account removes its refresh tokens, so the token alone can't
        // say whose it was. The client sends its stored user id as a hint; if that
        // account was deleted, say so, and the device wipes its local copy.
        const hint = req.body?.userId;
        if (typeof hint === 'string' && /^[0-9a-f-]{36}$/i.test(hint) && (await wasDeleted(hint))) {
          return reply.code(401).send({ error: 'account deleted', code: ACCOUNT_DELETED });
        }
        return reply.code(401).send({ error: 'invalid refresh token' });
      }
      const { user_id, device_name } = r.rows[0];
      const userRes = await pool.query<{ id: string; email: string }>(
        'select id, email from users where id = $1',
        [user_id],
      );
      if (!userRes.rowCount || !userRes.rows[0]) {
        return reply.code(401).send({ error: 'user not found' });
      }
      const { id, email } = userRes.rows[0];
      return issueTokens(id, email, device_name);
    },
  );

  app.post<{ Body: { refreshToken?: string } }>(
    '/logout',
    async (req) => {
      const token = req.body?.refreshToken;
      if (typeof token === 'string') {
        await pool.query('delete from refresh_tokens where token_hash = $1', [sha256(token)]);
      }
      return { ok: true };
    },
  );
};

export default authRoutes;

  const present = new Map(tables.rows.map((r) => [r.tablename, r.rowsecurity]));
  const missing = EXPECTED_TABLES.filter((t) => !present.has(t));
  if (missing.length) bad(`missing tables: ${missing.join(', ')} — run \`pnpm --filter server db:migrate\``);
  else ok(`schema present (${EXPECTED_TABLES.length} core tables)`);

  const migrations = await pool
    .query<{ name: string }>('select name from pgmigrations order by run_on')
    .catch(() => ({ rows: [] as { name: string }[] }));
  if (migrations.rows.length) ok(`migrations applied: ${migrations.rows.map((m) => m.name).join(', ')}`);
  else bad('no migrations recorded — run `pnpm --filter server db:migrate`');

  const hasApiRoles = await pool.query(`select 1 from pg_roles where rolname = 'anon'`);
  if (hasApiRoles.rowCount) {
    const open = [...present].filter(([, rls]) => !rls).map(([t]) => t);
    if (open.length) bad(`RLS disabled (exposed via Supabase REST API): ${open.join(', ')}`);
    else ok('all public tables have RLS enabled (Supabase REST API cannot read them)');
    const grants = await pool.query<{ table_name: string }>(
      `select distinct table_name from information_schema.role_table_grants
        where table_schema = 'public' and grantee in ('anon', 'authenticated')`,
    );
    if (grants.rowCount) bad(`anon/authenticated still have grants on: ${grants.rows.map((g) => g.table_name).join(', ')}`);
    else ok('anon/authenticated roles have no table privileges');
  }

  // LISTEN/NOTIFY round trip over the same path the SSE endpoint uses.
  const hub = new NotifyHub();
  const channel = `folio_check_${randomUUID()}`;
  try {
    let resolveReceived!: () => void;
    const received = new Promise<void>((resolve) => (resolveReceived = resolve));
    // Wait until LISTEN is in place, then NOTIFY from a different connection,
    // exactly as a sync push from another device would.
    await hub.subscribe(channel, () => resolveReceived());
    await pool.query('select pg_notify($1, $2)', [channel, '']);
    await Promise.race([
      received,
      new Promise((_, reject) => setTimeout(() => reject(new Error('timed out after 10s')), 10_000)),
    ]);
    ok('LISTEN/NOTIFY works (live sync between devices)');
  } catch (err) {
    bad(
      `LISTEN/NOTIFY failed (${(err as Error).message}). On Supabase use the session pooler ` +
        '(port 5432) or the direct connection, not the transaction pooler (port 6543).',
    );
  } finally {
    await hub.close();
  }
}

main()
  .catch((err) => {
    bad(`could not connect: ${err instanceof Error ? err.message : String(err)}`);
  })
  .finally(async () => {

async function calculateStorageUsed(userId: string): Promise<number> {
  const files = await storage.list(userId);
  return files.reduce((acc, f) => acc + f.size, 0);
}

export const accountRoutes: FastifyPluginAsync = async (app: FastifyInstance) => {
  // Get account info
  app.get(
    '/',
    { preHandler: app.auth },
    async (req, reply) => {
      const r = await pool.query<{ id: string; email: string; created_at: string }>(
        'select id, email, created_at from users where id = $1',
        [req.user.sub],
      );
      const user = r.rows[0];
      if (!user) {
        return reply.code(404).send({ error: 'user not found' });
      }

      const storageUsed = await calculateStorageUsed(user.id);
      return {
        id: user.id,
        email: user.email,
        storageUsed,
        createdAt: user.created_at,
      };
    },
  );

  // Change password
  app.patch<{ Body: { currentPassword?: string; newPassword?: string } }>(
    '/password',
    { preHandler: app.auth },
    async (req, reply) => {
      const { currentPassword, newPassword } = req.body || {};
      if (!currentPassword || !newPassword) {
        return reply.code(400).send({ error: 'currentPassword and newPassword required' });
      }
      if (newPassword.length < 8) {
        return reply.code(400).send({ error: 'newPassword must be at least 8 characters' });
      }

      const r = await pool.query<{ password_hash: string }>(
        'select password_hash from users where id = $1',
        [req.user.sub],
      );
      const user = r.rows[0];
      if (!user) {
        return reply.code(404).send({ error: 'user not found' });
      }

      const match = await bcrypt.compare(currentPassword, user.password_hash);
      if (!match) {
        return reply.code(401).send({ error: 'incorrect current password' });
      }

      const newHash = await bcrypt.hash(newPassword, 12);
      await pool.query('update users set password_hash = $1 where id = $2', [newHash, req.user.sub]);

      return { ok: true };
    },
  );

  // List active sessions
  app.get(
    '/sessions',
    { preHandler: app.auth },
    async (req) => {
      const r = await pool.query<{
        token_hash: string;
        device_name: string | null;
        created_at: string;
        expires_at: string;
      }>(
        `select token_hash, device_name, created_at, expires_at
         from refresh_tokens
         where user_id = $1 and expires_at > now()
         order by created_at desc`,
        [req.user.sub],
      );
      return r.rows;
    },
  );

  // Revoke a specific session
  app.delete<{ Params: { tokenHash: string } }>(
    '/sessions/:tokenHash',
    { preHandler: app.auth },
    async (req) => {
      await pool.query(
        'delete from refresh_tokens where user_id = $1 and token_hash = $2',
        [req.user.sub, req.params.tokenHash],
      );
      return { ok: true };
    },
  );

  // Export all user data as a ZIP archive
  app.get(
    '/export',
    { preHandler: app.auth },
    async (req, reply) => {
      const userId = req.user.sub;
      const exportData: Record<string, unknown[]> = {};

      for (const table of Object.keys(SYNC_TABLES) as SyncTableName[]) {
        const r = await pool.query(
          `select * from ${table} where user_id = $1 and deleted_at is null order by updated_at desc`,
          [userId],
        );
        exportData[table] = r.rows.map(({ user_id: _, ...rest }) => rest);
      }

      // Collect file manifest
      const filesManifest = await storage.list(userId);

      reply.raw.writeHead(200, {
        'Content-Type': 'application/zip',
        'Content-Disposition': `attachment; filename="verso-export-${Date.now()}.zip"`,
        'Cache-Control': 'no-cache',
      });

      const archive = archiver('zip', { zlib: { level: 6 } });
      archive.pipe(reply.raw);

      // Append metadata
      archive.append(
        JSON.stringify(
          {
            userId,
            email: req.user.email,
            exportedAt: new Date().toISOString(),
            tables: Object.keys(exportData),
            fileCount: filesManifest.length,
          },
          null,
          2,
        ),
        { name: 'manifest.json' },
      );

      // Append table JSONs
      for (const [table, rows] of Object.entries(exportData)) {
        archive.append(JSON.stringify(rows, null, 2), { name: `tables/${table}.json` });
      }

      // Append files list
      archive.append(JSON.stringify(filesManifest, null, 2), { name: 'files.json' });

      await archive.finalize();
    },
  );

  // Delete account cascade
  app.delete(
    '/',
    { preHandler: app.auth },
    async (req) => {
      const userId = req.user.sub;
      // Tombstone + delete in one transaction (rows cascade via foreign keys).
      const client = await pool.connect();
      try {
        await client.query('begin');
        await client.query(
          'insert into deleted_users (id) values ($1) on conflict (id) do nothing',
          [userId],
        );
        await client.query('delete from users where id = $1', [userId]);
        await client.query('commit');
      } catch (e) {
        await client.query('rollback');
        throw e;
      } finally {
        client.release();
      }
      forgetUser(userId);

      // Wake this user's other signed-in devices: their next sync gets
      // "account deleted" and wipes their copy instead of re-uploading it.
      await pool.query('select pg_notify($1, $2)', [`sync_${userId}`, '']).catch(() => {});

      // The account is already gone; a storage hiccup must not make the client
      // think deletion failed. Retry file removal in the background.
      void (async () => {
        for (let attempt = 0; attempt < 5; attempt++) {
          try {
            await storage.deleteAll(userId);
            return;
          } catch (err) {
            req.log.warn({ err, attempt }, 'deleting account files failed; retrying');
            await new Promise((r) => setTimeout(r, 2000 * 2 ** attempt));
          }
        }
        req.log.error({ userId }, 'could not delete account files');
      })();

      return { ok: true };
    },
  );
};

import fs from 'node:fs';
import fsp from 'node:fs/promises';
import path from 'node:path';
import {
  S3Client,
  PutObjectCommand,
  GetObjectCommand,
  HeadObjectCommand,

  DeleteObjectCommand,
  DeleteObjectsCommand,
  ListObjectsV2Command,
} from '@aws-sdk/client-s3';

export const HASH_RE = /^[a-f0-9]{64}$/;

/**
 * The storage service can't be reached (e.g. the PC running s3rver is off or the
 * ngrok tunnel is closed). Routes answer 503 "storage_offline" for this instead
 * of claiming the file doesn't exist.
 */
export class StorageOfflineError extends Error {
  constructor(cause?: unknown) {
    super('Book storage is offline', { cause });
    this.name = 'StorageOfflineError';
  }
}

type AwsError = { name?: string; code?: string; message?: string; $metadata?: { httpStatusCode?: number } };

/** Network-level failure or a gateway/tunnel error rather than an S3 answer. */
function isUnreachable(err: unknown): boolean {
  const e = err as AwsError;
  const status = e.$metadata?.httpStatusCode;
  if (status === 502 || status === 503 || status === 504) return true;
  const text = `${e.name ?? ''} ${e.code ?? ''} ${e.message ?? ''}`;
  return /ECONNREFUSED|ENOTFOUND|ETIMEDOUT|ECONNRESET|EAI_AGAIN|socket hang up|TimeoutError|ERR_NGROK|Unexpected end|Deserialization|char '<'/i.test(text);
}

export interface FileItem {
  hash: string;
  size: number;
}

export interface StorageDriver {
  has(userId: string, hash: string): Promise<boolean>;
  get(userId: string, hash: string): Promise<{ stream: NodeJS.ReadableStream; size: number } | null>;
  put(userId: string, hash: string, localFilePath: string, size: number): Promise<void>;
  delete(userId: string, hash: string): Promise<void>;
  list(userId: string): Promise<FileItem[]>;
  deleteAll(userId: string): Promise<void>;
  /** Throws StorageOfflineError if the backing store can't be reached. */
  ping(): Promise<void>;
}

class LocalStorageDriver implements StorageDriver {
  private root: string;

  constructor(root = process.env.FILES_DIR || './data/files') {
    this.root = root;
  }

  private userDir(userId: string): string {
    return path.join(this.root, userId);
  }

  private filePath(userId: string, hash: string): string {
    return path.join(this.userDir(userId), hash);
  }

  async has(userId: string, hash: string): Promise<boolean> {
    try {
      await fsp.access(this.filePath(userId, hash));
      return true;
    } catch {
      return false;
    }
  }

  async get(userId: string, hash: string): Promise<{ stream: NodeJS.ReadableStream; size: number } | null> {
    const p = this.filePath(userId, hash);
    try {
      const stat = await fsp.stat(p);
      return {
        stream: fs.createReadStream(p),
        size: stat.size,
      };
    } catch {
      return null;
    }
  }

  async put(userId: string, hash: string, localFilePath: string): Promise<void> {
    const dir = this.userDir(userId);
    await fsp.mkdir(dir, { recursive: true });
    const final = this.filePath(userId, hash);
    if (localFilePath !== final) {
      await fsp.rename(localFilePath, final);
    }
  }

  async delete(userId: string, hash: string): Promise<void> {
    await fsp.rm(this.filePath(userId, hash), { force: true });
  }

  async list(userId: string): Promise<FileItem[]> {
    const dir = this.userDir(userId);
    try {
      const entries = await fsp.readdir(dir, { withFileTypes: true });
      const files: FileItem[] = [];
      for (const entry of entries) {
        if (entry.isFile() && HASH_RE.test(entry.name)) {
          const stat = await fsp.stat(path.join(dir, entry.name));
          files.push({ hash: entry.name, size: stat.size });
        }
      }
      return files;
    } catch {
      return [];
    }
  }

  async deleteAll(userId: string): Promise<void> {
    await fsp.rm(this.userDir(userId), { recursive: true, force: true });
  }

  async ping(): Promise<void> {
    await fsp.mkdir(this.root, { recursive: true });
  }
}

class R2StorageDriver implements StorageDriver {
  private client: S3Client;
  private bucket: string;

  constructor(config: { endpoint: string; accessKeyId: string; secretAccessKey: string; bucket: string; region?: string }) {
    this.bucket = config.bucket;
    this.client = new S3Client({
      region: config.region || 'auto',
      endpoint: config.endpoint,
      forcePathStyle: true,
      credentials: {
        accessKeyId: config.accessKeyId,
        secretAccessKey: config.secretAccessKey,
      },
    });
  }

  private key(userId: string, hash: string): string {
    return `${userId}/${hash}`;
  }

  private reachableUntil = 0;

  /**
   * Confirms the bucket answers. A tunnel that's down (ngrok) replies 404 to
   * every request, which on its own looks exactly like "file not found", so a
   * not-found result is only trusted once the bucket itself is reachable.
   */
  async ping(): Promise<void> {
    if (Date.now() < this.reachableUntil) return;
    try {
      await this.client.send(new HeadBucketCommand({ Bucket: this.bucket }));
      this.reachableUntil = Date.now() + 30_000;
    } catch (err) {
      throw new StorageOfflineError(err);
    }
  }

  private async rethrow(err: unknown): Promise<never> {
    if (isUnreachable(err)) throw new StorageOfflineError(err);
    throw err;
  }

  async has(userId: string, hash: string): Promise<boolean> {
    try {
      await this.client.send(
        new HeadObjectCommand({
          Bucket: this.bucket,
          Key: this.key(userId, hash),
        }),
      );
      return true;
    } catch (err: unknown) {
      const e = err as AwsError;
      if (e.name === 'NotFound' || e.name === 'NoSuchKey' || e.$metadata?.httpStatusCode === 404) {
        await this.ping();
        return false;
      }
      return this.rethrow(err);
    }
  }

  async get(userId: string, hash: string): Promise<{ stream: NodeJS.ReadableStream; size: number } | null> {
    try {
      const res = await this.client.send(
        new GetObjectCommand({
          Bucket: this.bucket,
          Key: this.key(userId, hash),
        }),
      );
      if (!res.Body) return null;
      return {
        stream: res.Body as NodeJS.ReadableStream,
        size: res.ContentLength ?? 0,
      };
    } catch (err: unknown) {
      const e = err as AwsError;
      if (e.name === 'NotFound' || e.name === 'NoSuchKey' || e.$metadata?.httpStatusCode === 404) {
        await this.ping();
        return null;
      }
      return this.rethrow(err);
    }
  }

  async put(userId: string, hash: string, localFilePath: string, size: number): Promise<void> {
    const fileStream = fs.createReadStream(localFilePath);
    try {
      await this.client.send(
        new PutObjectCommand({
          Bucket: this.bucket,
          Key: this.key(userId, hash),
          Body: fileStream,
          ContentLength: size,
          ContentType: 'application/octet-stream',
        }),
      );
    } catch (err) {
      await this.rethrow(err);
    } finally {
      await fsp.rm(localFilePath, { force: true });
    }
  }

  async delete(userId: string, hash: string): Promise<void> {
    await this.client
      .send(new DeleteObjectCommand({ Bucket: this.bucket, Key: this.key(userId, hash) }))
      .catch((err) => this.rethrow(err));
  }

  /** All object keys under a prefix. ListObjectsV2 returns at most 1000 per page. */
  private async listAll(prefix: string): Promise<{ Key: string; Size: number }[]> {
    const out: { Key: string; Size: number }[] = [];
    let token: string | undefined;
    do {
      const res = await this.client
        .send(new ListObjectsV2Command({ Bucket: this.bucket, Prefix: prefix, ContinuationToken: token }))
        .catch(async (err) => {
          // A 404 for the whole listing is a closed tunnel (or a missing bucket).
          if ((err as AwsError).$metadata?.httpStatusCode === 404) await this.ping();
          return this.rethrow(err);
        });
      for (const obj of res.Contents || []) {
        if (obj.Key) out.push({ Key: obj.Key, Size: obj.Size ?? 0 });
      }
      token = res.IsTruncated ? res.NextContinuationToken : undefined;
    } while (token);
    return out;
  }

  async list(userId: string): Promise<FileItem[]> {
    const prefix = `${userId}/`;
    const files: FileItem[] = [];
    for (const obj of await this.listAll(prefix)) {
      const hash = obj.Key.slice(prefix.length);
      if (HASH_RE.test(hash)) files.push({ hash, size: obj.Size });
    }
    return files;
  }

  async deleteAll(userId: string): Promise<void> {
    const keys = (await this.listAll(`${userId}/`)).map((o) => ({ Key: o.Key }));
    // DeleteObjects accepts at most 1000 keys per request.
    for (let i = 0; i < keys.length; i += 1000) {
      await this.client.send(
        new DeleteObjectsCommand({ Bucket: this.bucket, Delete: { Objects: keys.slice(i, i + 1000) } }),
      );
    }
  }
}

function createStorageDriver(): StorageDriver {
  const endpoint =
    process.env.R2_ENDPOINT ||
    (process.env.R2_ACCOUNT_ID ? `https://${process.env.R2_ACCOUNT_ID}.r2.cloudflarestorage.com` : undefined);
  const accessKeyId = process.env.R2_ACCESS_KEY_ID;
  const secretAccessKey = process.env.R2_SECRET_ACCESS_KEY;
  const bucket = process.env.R2_BUCKET_NAME;

  if (endpoint && accessKeyId && secretAccessKey && bucket) {
    return new R2StorageDriver({
      endpoint,
      accessKeyId,
      secretAccessKey,
      bucket,
      region: process.env.R2_REGION,
    });
  }

  return new LocalStorageDriver();
}

import type { FastifyInstance, FastifyPluginAsync } from 'fastify';
import type { PoolClient } from 'pg';
import { pool } from './db.js';
import { getNotifyHub } from './notify.js';
import { getInstanceId, lockUserWrites } from './integrity.js';
import {
  SYNC_TABLES,
  type SyncTableName,
  type SyncPushResponse,
  type SyncPullResponse,
  type EventsTokenResponse,
} from '@verso/shared';
import './types.js';

const JSON_COLS = new Set(['meta', 'data']);
const MAX_PUSH_ROWS = 5000;
const PULL_LIMIT = 1000;

function validRow(row: unknown): row is { id: string; updated_at: number; deleted_at?: number | null; [key: string]: unknown } {
  if (!row || typeof row !== 'object') return false;
  const r = row as Record<string, unknown>;
  return (
    typeof r.id === 'string' &&
    r.id.length > 0 &&
    r.id.length <= 128 &&
    Number.isFinite(Number(r.updated_at))
  );
}

async function upsert(
  client: PoolClient,
  table: SyncTableName,
  userId: string,
  row: Record<string, unknown>,
): Promise<boolean> {
  const incomingUpdated = Number(row.updated_at);
  const existing = await client.query<{ updated_at: number }>(
    `select updated_at from ${table} where user_id = $1 and id = $2`,
    [userId, row.id],
  );

  if (existing.rowCount && existing.rows[0]) {
    if (incomingUpdated <= Number(existing.rows[0].updated_at)) {
      return false;
    }
  }

  const fields = SYNC_TABLES[table];
  const cols = ['user_id', 'id', 'updated_at', 'deleted_at', ...fields];
  const vals: unknown[] = [
    userId,
    row.id,
    incomingUpdated,
    row.deleted_at == null ? null : Number(row.deleted_at),
  ];

  for (const field of fields) {
    const val = row[field];
    if (JSON_COLS.has(field)) {
      vals.push(val != null ? JSON.stringify(val) : '{}');
    } else if (field === 'read') {
      vals.push(val != null ? Boolean(val) : false);
    } else if (field === 'sort_order') {
      vals.push(val != null ? Number(val) : 0);
    } else {
      vals.push(val ?? null);
    }
  }

  const ph = vals.map((_, i) => `$${i + 1}`).join(',');
  const setCols = cols.slice(2).map((c) => `${c} = excluded.${c}`);
  setCols.push("server_seq = nextval('sync_seq')");
  const set = setCols.join(',');
  const query = `
    insert into ${table} (${cols.join(',')}) values (${ph})
    on conflict (user_id, id) do update set ${set}
    where ${table}.updated_at < excluded.updated_at
  `;

  await client.query(query, vals);
  return true;
}

export const syncRoutes: FastifyPluginAsync = async (app: FastifyInstance) => {
  // Push changes from client
  app.post<{ Body: { changes?: Record<string, unknown[]> } }>(
    '/push',
    { preHandler: app.auth },
    async (req, reply) => {
      const changes = req.body?.changes;
      if (!changes || typeof changes !== 'object') {
        return reply.code(400).send({ error: 'changes object required' });
      }

      const total = Object.values(changes).reduce(
        (n, r) => n + (Array.isArray(r) ? r.length : 0),
        0,
      );
      if (total > MAX_PUSH_ROWS) {
        return reply.code(413).send({ error: `max ${MAX_PUSH_ROWS} rows per push` });
      }

      const client = await pool.connect();
      let accepted = 0;
      let stale = 0;
      let invalid = 0;
      const staleIds: Partial<Record<SyncTableName, string[]>> = {};
      const serverRows: Partial<Record<SyncTableName, Record<string, unknown>[]>> = {};

      try {
        await client.query('begin');
        await lockUserWrites(client, req.user.sub);
        for (const [tbl, rows] of Object.entries(changes)) {
          const table = tbl as SyncTableName;
          if (!SYNC_TABLES[table] || !Array.isArray(rows)) continue;

          for (const row of rows) {
            if (!validRow(row)) {
              invalid++;
              continue;
            }
            const isAccepted = await upsert(client, table, req.user.sub, row);
            if (isAccepted) {
              accepted++;
            } else {
              stale++;
              if (!staleIds[table]) {
                staleIds[table] = [];
              }
              staleIds[table]!.push(row.id as string);
              // Send back the winning copy so the client converges on it.
              const current = await client.query<Record<string, unknown>>(
                `select * from ${table} where user_id = $1 and id = $2`,
                [req.user.sub, row.id],
              );
              if (current.rows[0]) {
                const { user_id: _u, ...rest } = current.rows[0];
                (serverRows[table] ??= []).push(rest);
              }
            }
          }
        }
        await client.query('commit');
      } catch (e) {
        await client.query('rollback');
        throw e;
      } finally {
        client.release();
      }

      const response: SyncPushResponse = {
        accepted,
        stale,
        invalid,
        staleIds,
        serverRows,
        serverTime: Date.now(),
      };
      return response;
    },
  );

  // Pull changes from server since seq
  app.get<{ Querystring: { since?: string } }>(
    '/pull',
    { preHandler: app.auth },
    async (req): Promise<SyncPullResponse> => {
      const since = Math.max(0, Number(req.query.since) || 0);
      const fetched: Partial<Record<SyncTableName, Record<string, unknown>[]>> = {};
      let ceiling = Infinity;

      for (const table of Object.keys(SYNC_TABLES) as SyncTableName[]) {
        const r = await pool.query<Record<string, unknown>>(
          `select * from ${table} where user_id = $1 and server_seq > $2
           order by server_seq limit $3`,
          [req.user.sub, since, PULL_LIMIT],
        );
        fetched[table] = r.rows;
        if (r.rows.length === PULL_LIMIT) {
          const lastSeq = Number(r.rows[r.rows.length - 1].server_seq);
          ceiling = Math.min(ceiling, lastSeq);
        }
      }

      const hasMore = ceiling !== Infinity;
      const changes: Partial<Record<SyncTableName, Record<string, unknown>[]>> = {};
      let cursor = since;

      for (const [tbl, rows] of Object.entries(fetched)) {
        const table = tbl as SyncTableName;
        if (!rows) continue;
        changes[table] = rows
          .filter((r) => Number(r.server_seq) <= ceiling)
          .map(({ user_id: _, ...rest }) => rest);

        for (const r of changes[table]!) {
          cursor = Math.max(cursor, Number(r.server_seq));
        }
      }

      if (hasMore) cursor = ceiling;

      return {
        changes,
        cursor,
        hasMore,
        serverTime: Date.now(),
        instanceId: await getInstanceId(),
      };
    },
  );

  // Issue 5-minute SSE token
  app.post(
    '/events-token',
    { preHandler: app.auth },
    async (req): Promise<EventsTokenResponse> => {
      const token = app.jwt.sign(
        { sub: req.user.sub, email: req.user.email, type: 'events' },
        { expiresIn: '5m' },
      );
      return { token, expiresIn: 300 };
    },
  );

  // Server-Sent Events stream for live updates
  app.get<{ Querystring: { token?: string } }>(
    '/events',
    async (req, reply) => {
      const token = req.query?.token;
      if (!token || typeof token !== 'string') {
        return reply.code(401).send({ error: 'token required' });
      }

      let userId: string;
      try {
        const decoded = app.jwt.verify<{ sub: string; type?: string }>(token);
        if (decoded.type !== 'events' || !decoded.sub) {
          return reply.code(401).send({ error: 'invalid events token' });
        }
        userId = decoded.sub;
      } catch {
        return reply.code(401).send({ error: 'invalid or expired events token' });
      }

      reply.raw.writeHead(200, {
        'Content-Type': 'text/event-stream',
        'Cache-Control': 'no-cache, no-transform',
        Connection: 'keep-alive',
        'X-Accel-Buffering': 'no',
      });
      reply.raw.flushHeaders?.();

      // Send initial comment to establish stream
      reply.raw.write(': stream opened\n\n');

      const channel = `sync_${userId}`;

      let debounceTimer: NodeJS.Timeout | null = null;
      const onNotification = () => {
        if (debounceTimer) clearTimeout(debounceTimer);
        debounceTimer = setTimeout(() => {
          if (!reply.raw.writableEnded) {
            reply.raw.write('event: changed\ndata: {}\n\n');
          }
        }, 500);
      };

      // All streams share one LISTEN connection instead of each holding a pooled client.
      const unsubscribe = await getNotifyHub().subscribe(channel, onNotification);

      // Heartbeat comment every 25s
      const heartbeat = setInterval(() => {
        if (!reply.raw.writableEnded) {
          reply.raw.write(': keep-alive\n\n');
        }
      }, 25000);

      req.raw.on('close', () => {
        if (debounceTimer) clearTimeout(debounceTimer);
        clearInterval(heartbeat);
        void unsubscribe();
      });
    },
  );
};

export default syncRoutes;

export async function createServer(): Promise<FastifyInstance> {
  const jwtSecret = process.env.JWT_SECRET || 'dev_secret_must_be_at_least_32_characters_long_for_hs256';

  const app = Fastify({
    logger: process.env.NODE_ENV !== 'test',
    bodyLimit: 20 * 1024 * 1024,
    trustProxy: true,
  });

  const origins = process.env.CORS_ORIGINS || '*';
  await app.register(cors, {
    origin: origins === '*' ? true : origins.split(',').map((s) => s.trim()),
    methods: ['GET', 'HEAD', 'POST', 'PUT', 'PATCH', 'DELETE'],
  });

  if (process.env.NODE_ENV !== 'test') {
    await app.register(rateLimit, {
      max: 300,
      timeWindow: '1 minute',
    });
  }

  await app.register(jwt, {
    secret: jwtSecret,
  });

  app.decorate('auth', async (req, reply) => {
    try {
      await req.jwtVerify();
    } catch {
      return reply.code(401).send({ error: 'unauthorized' });
    }
    // Access tokens outlive account deletion by up to their TTL. Reject them, and
    // tell the device the account is gone so it wipes its copy rather than
    // pushing it back (which also failed on the users foreign key).
    if (!(await userExists(req.user.sub))) {
      const deleted = await wasDeleted(req.user.sub);
      return reply
        .code(401)
        .send({ error: deleted ? 'account deleted' : 'unauthorized', code: deleted ? ACCOUNT_DELETED : undefined });
    }
  });

  // Storage offline (PC with s3rver off / tunnel closed): a clear 503 instead of a
  // generic 500, or "not found" for files that do exist.
  app.setErrorHandler((err, _req, reply) => {
    if (err instanceof StorageOfflineError) {
      return reply.code(503).send({ error: 'book storage is offline', code: STORAGE_OFFLINE });
    }
    return reply.send(err);
  });

  // Health endpoint checks database connectivity
  app.get('/health', async () => {
    await pool.query('select 1');
    return { ok: true };
  });

  // Is the book file storage reachable? (open in a browser to check your tunnel)
  app.get('/health/storage', async (_req, reply) => {
    try {
      await storage.ping();
      return { ok: true };
    } catch {
      return reply.code(503).send({ ok: false, code: STORAGE_OFFLINE });
    }
  });

  // Me endpoint
  app.get('/me', { preHandler: app.auth }, async (req) => ({
    id: req.user.sub,
    email: req.user.email,
  }));

  // Feature routes
  await app.register(authRoutes, { prefix: '/auth' });
  await app.register(syncRoutes, { prefix: '/sync' });
  await app.register(fileRoutes, { prefix: '/files' });
  await app.register(proxyRoutes, { prefix: '/proxy' });
  await app.register(accountRoutes, { prefix: '/account' });

  return app;
}

import type { FastifyInstance, FastifyPluginAsync } from 'fastify';
import { Readable } from 'node:stream';
import { pool } from './db.js';
import './types.js';

const STATIC_ALLOWED_HOSTS = new Set([
  'gutendex.com',
  'www.gutenberg.org',
  'gutenberg.org',
  'standardebooks.org',
  'api.mangadex.org',
  'uploads.mangadex.org',
  'openlibrary.org',
  'covers.openlibrary.org',
  'archive.org',
  'mangapill.com',
  'cdn.readdetectiveconan.com',
  'raw.githubusercontent.com',
  'cdn.jsdelivr.net',
  'localhost',
  '127.0.0.1',
]);

function isStaticAllowedHost(hostname: string): boolean {
  const host = hostname.toLowerCase();
  if (STATIC_ALLOWED_HOSTS.has(host)) return true;
  if (host.endsWith('.mangadex.network')) return true;
  if (host.endsWith('.archive.org')) return true;
  if (host.endsWith('readdetectiveconan.com')) return true;
  if (host.endsWith('mangapill.com')) return true;
  return false;
}

// In-memory token bucket rate limiters per user
interface RateBucket {
  tokens: number;
  lastRefill: number;
  maxTokens: number;
  refillRatePerSec: number;
}

const rateBuckets = new Map<string, RateBucket>();

function checkRateLimit(key: string, maxTokens: number, refillRatePerSec: number): boolean {
  const now = Date.now();
  let bucket = rateBuckets.get(key);
  if (!bucket) {
    bucket = { tokens: maxTokens, lastRefill: now, maxTokens, refillRatePerSec };
    rateBuckets.set(key, bucket);
  }

  const elapsedSec = (now - bucket.lastRefill) / 1000;
  bucket.tokens = Math.min(bucket.maxTokens, bucket.tokens + elapsedSec * bucket.refillRatePerSec);
  bucket.lastRefill = now;

  if (bucket.tokens >= 1) {
    bucket.tokens -= 1;
    return true;
  }
  return false;
}

// Clean up old rate buckets periodically
setInterval(() => {
  const now = Date.now();
  for (const [key, bucket] of rateBuckets.entries()) {
    if (now - bucket.lastRefill > 60000) {
      rateBuckets.delete(key);
    }
  }
}, 60000);

export const proxyRoutes: FastifyPluginAsync = async (app: FastifyInstance) => {
  const handler = async (
    req: import('fastify').FastifyRequest<{ Querystring: { url?: string } }>,
    reply: import('fastify').FastifyReply,
  ) => {
    const rawUrl = req.query?.url;
    if (!rawUrl || typeof rawUrl !== 'string') {
      return reply.code(400).send({ error: 'url parameter required' });
    }

    let parsed: URL;
    try {
      parsed = new URL(rawUrl);
    } catch {
      return reply.code(400).send({ error: 'invalid url' });
    }

    if (parsed.protocol !== 'http:' && parsed.protocol !== 'https:') {
      return reply.code(400).send({ error: 'only http and https supported' });
    }

    const hostname = parsed.hostname.toLowerCase();

    // Try verifying JWT if authorization header is provided
    let userId: string | null = null;
    if (req.headers.authorization) {
      try {
        await req.jwtVerify();
        userId = req.user.sub;
      } catch {
        // ignore invalid token for static allowed hosts, but don't grant user permissions
      }
    }

    // Rate limit checks:
    const rateLimitKey = userId ? `user:${userId}` : `ip:${req.ip}`;
    if (!checkRateLimit(`overall:${rateLimitKey}`, 10, 10)) {
      return reply.code(429).send({ error: 'overall proxy rate limit exceeded (10 req/s)' });
    }

    // 4 req/s for api.mangadex.org
    if (hostname === 'api.mangadex.org') {
      if (!checkRateLimit(`mangadex:${rateLimitKey}`, 4, 4)) {
        return reply.code(429).send({ error: 'mangadex proxy rate limit exceeded (4 req/s)' });
      }
    }

    // Check allowlist: static hosts first
    let isAllowed = isStaticAllowedHost(hostname);
    let isUserSourceHost = false;

    // Check user source hosts if not static allowed and user is authenticated
    if (!isAllowed && userId) {
      const sourcesRes = await pool.query<{ url: string }>(
        'select url from sources where user_id = $1 and deleted_at is null',
        [userId],
      );
      for (const row of sourcesRes.rows) {
        try {
          const sourceUrl = new URL(row.url);
          if (sourceUrl.hostname.toLowerCase() === hostname) {
            isAllowed = true;
            isUserSourceHost = true;
            break;
          }
        } catch {
          // ignore malformed source url
        }
      }
    }

    if (!isAllowed) {
      return reply.code(403).send({ error: `host not allowed: ${hostname}` });
    }

    // Construct forward headers
    const forwardHeaders: Record<string, string> = {
      'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36',
    };

    if (hostname.endsWith('readdetectiveconan.com') || hostname.endsWith('mangapill.com')) {
      forwardHeaders['Referer'] = 'https://mangapill.com/';
    }

    if (req.headers.accept) {
      forwardHeaders['Accept'] = req.headers.accept;
    }
    if (req.headers.range) {
      forwardHeaders['Range'] = req.headers.range;
    }
    // Forward Content-Type for POST/PUT requests (e.g. GraphQL mutations)
    if (req.headers['content-type']) {
      forwardHeaders['Content-Type'] = req.headers['content-type'];
    }
    // Forward Authorization ONLY for user source hosts
    if (isUserSourceHost && req.headers.authorization) {
      forwardHeaders['Authorization'] = req.headers.authorization;
    }

    // Determine request body for POST/PUT
    const method = req.method as string;
    const fetchInit: RequestInit = {
      method,
      headers: forwardHeaders,
      redirect: 'follow',
    };
    if (method === 'POST' || method === 'PUT') {
      const rawBody = req.body;
      fetchInit.body = typeof rawBody === 'string' ? rawBody : JSON.stringify(rawBody);
    }

    let upstreamRes: Response;
    try {
      upstreamRes = await fetch(parsed.toString(), fetchInit);
    } catch (err: unknown) {
      const error = err as Error;
      return reply.code(502).send({ error: 'upstream fetch failed', message: error.message });
    }

    reply.code(upstreamRes.status);

    // Forward relevant headers
    const forwardableHeaders = [
      'content-type',
      'content-length',
      'content-range',
      'accept-ranges',
      'etag',
      'last-modified',
      'cache-control',
    ];

    for (const h of forwardableHeaders) {
      const val = upstreamRes.headers.get(h);
      if (val) {
        reply.header(h, val);
      }
    }

    if (!upstreamRes.body) {
      return reply.send();
    }

    // Stream the body back
    const nodeStream = Readable.fromWeb(upstreamRes.body as import('node:stream/web').ReadableStream);
    return reply.send(nodeStream);
  };

  app.get<{ Querystring: { url?: string } }>('/', handler);
  app.post<{ Querystring: { url?: string } }>('/', handler);
};

export default proxyRoutes;
